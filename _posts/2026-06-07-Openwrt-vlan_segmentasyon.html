VLAN ile ev ağını segmentlere ayırmak

Tüm cihazlar aynı ağda olduğunda akıllı TV'niz NAS'ınıza, iot cihazlarınız iş bilgisayarınıza erişebilir. VLAN bu erişim kontrolünü switch yazılımı seviyesinde çözer.

802.1Q VLAN (Virtual LAN), tek bir fiziksel ağ altyapısı üzerinde birden fazla izole ağ segmenti oluşturmanın standart yoludur. Her VLAN, farklı bir broadcast domain'dir: bir VLAN'daki cihaz diğer VLAN'ı göremez, duyamaz, tarayamaz. Aralarındaki trafik sadece router üzerinden ve sadece izin verilen kurallara göre geçebilir.

Neden VLAN gerekli?

  • ✗IoT cihazları (kamera, termostat, akıllı ampul) ele geçirildiğinde tüm ağa erişim sağlar
  • ✗Misafir Wi-Fi aynı subnet'teyse misafirler yerel NAS ve yazıcılara erişebilir
  • ✗Bir cihazdan broadcast storm veya ARP flood tüm ağı etkiler

OpenWrt yapılandırması

Ağ arayüzleri — /etc/config/network

# Ana LAN — VLAN 10
config interface 'lan'
    option device   'br-lan.10'
    option proto    'static'
    option ipaddr   '192.168.10.1'
    option netmask  '255.255.255.0'

# IoT segmenti — VLAN 20
config interface 'iot'
    option device   'br-lan.20'
    option proto    'static'
    option ipaddr   '192.168.20.1'
    option netmask  '255.255.255.0'

# Misafir segmenti — VLAN 30
config interface 'guest'
    option device   'br-lan.30'
    option proto    'static'
    option ipaddr   '192.168.30.1'
    option netmask  '255.255.255.0'

# Yönetim segmenti — VLAN 99
config interface 'mgmt'
    option device   'br-lan.99'
    option proto    'static'
    option ipaddr   '192.168.99.1'
    option netmask  '255.255.255.0'

# Bridge VLAN filtreleme
config bridge-vlan
    option device   'br-lan'
    option vlan     '10'
    list ports      'lan1:u*'  # untagged — PC, NAS portları
    list ports      'lan4:t'   # tagged — uplink/trunk

config bridge-vlan
    option device   'br-lan'
    option vlan     '20'
    list ports      'lan2:u*'  # untagged — IoT portları
    list ports      'lan4:t'

config bridge-vlan
    option device   'br-lan'
    option vlan     '30'
    list ports      'lan3:u*'  # untagged — misafir AP
    list ports      'lan4:t'

config bridge-vlan
    option device   'br-lan'
    option vlan     '99'
    list ports      'lan4:t'   # sadece trunk

DHCP — /etc/config/dhcp

config dhcp 'iot'
    option interface  'iot'
    option start      '100'
    option limit      '50'
    option leasetime  '12h'

config dhcp 'guest'
    option interface  'guest'
    option start      '100'
    option limit      '50'
    option leasetime  '2h'

config dhcp 'mgmt'
    option interface  'mgmt'
    option ignore     '1'  # yönetim DHCP vermez, statik IP

Wi-Fi SSID → VLAN eşlemesi — /etc/config/wireless

config wifi-iface 'wifinet_iot'
    option device    'radio0'
    option mode      'ap'
    option ssid      'Ev-IoT'
    option network   'iot'
    option encryption 'psk2'
    option key       'guclu_sifre'

config wifi-iface 'wifinet_guest'
    option device    'radio0'
    option mode      'ap'
    option ssid      'Misafir-WiFi'
    option network   'guest'
    option encryption 'psk2'
    option key       'misafir_sifre'
    option isolate   '1'  # AP client isolation

Firewall kuralları — nftables

DENYIoT → LAN erişimi
IoT cihazlarının ana LAN'a erişimini tamamen engeller.
iifname "iot" oifname "lan" drop
DENYMisafir → LAN/IoT
Misafir ağından hem LAN hem IoT'a erişimi engeller.
iifname "guest" oifname {"lan","iot"} drop
ALLOWIoT → İnternet
IoT cihazları sadece internete çıkabilir, yerel ağa değil.
iifname "iot" oifname "wan" accept
ALLOWMisafir → HTTP/HTTPS
Misafirler sadece 80 ve 443 portlarına çıkabilir.
tcp dport {80,443} accept
DENYYönetim → dışarı
Yönetim VLAN'ı internete çıkamaz, sadece yerel erişim.
iifname "mgmt" oifname "wan" drop
ALLOWLAN → Yönetim
Yönetim arayüzlerine sadece ana LAN'dan erişilebilir.
iifname "lan" oifname "mgmt" accept 
# /etc/nftables.d/20-vlan-isolation.nft
table inet vlan_policy {
    chain forward {
        type filter hook forward priority 0; policy drop;

        # İzin verilen outbound: tüm VLAN → WAN
        oifname "wan" accept

        # Ana LAN tam erişim
        iifname "lan" accept

        # IoT: sadece WAN
        iifname "iot" oifname "wan" accept
        iifname "iot" drop

        # Misafir: sadece HTTP/HTTPS + DNS
        iifname "guest" oifname "wan" tcp dport {80, 443} accept
        iifname "guest" oifname "wan" udp dport 53 accept
        iifname "guest" drop

        # Established/related
        ct state established,related accept
    }
}
DNS sızıntısına dikkat: IoT ve misafir cihazlar kendi gateway'ine DNS sorgusu atar. Router'da SmartDNS veya dnsmasq bu sorguları yanıtlar. DoT upstream kullanıyorsanız IoT trafiği de şifreli DNS'den geçer.

Managed switch yapılandırması

# Port rolleri (TL-SG105E örneği):
Port 1 → Trunk  → Router uplink  (VLAN 10/20/30/99 tagged)
Port 2 → Access → VLAN 10        (PC, NAS — untagged)
Port 3 → Access → VLAN 20        (IoT cihazlar — untagged)
Port 4 → Access → VLAN 30        (Misafir AP — untagged)
Port 5 → Access → VLAN 99        (Switch yönetim — untagged)

# VLAN tablosu:
VLAN ID | Port 1 | Port 2 | Port 3 | Port 4 | Port 5
--------|--------|--------|--------|--------|-------
10      | Tagged | Untagg | -      | -      | -
20      | Tagged | -      | Untagg | -      | -
30      | Tagged | -      | -      | Untagg | -
99      | Tagged | -      | -      | -      | Untagg
Access port vs trunk port: Access port tek bir VLAN'a ait frame'leri tag'siz iletir. Trunk port birden fazla VLAN'ı 802.1Q tag'leriyle taşır. Router ve switch arasındaki link her zaman trunk olmalıdır.

Test ve doğrulama

  1. 1IoT cihazını VLAN 20 portuna bağla, ping 192.168.10.1 ile ana LAN gateway'ine ulaşamamalı.
  2. 2Aynı cihazdan curl https://example.com ile internete çıkabilmeli.
  3. 3Misafir Wi-Fi'dan nmap -sn 192.168.10.0/24 ile ana LAN'ı tarayamamalı.
  4. 4Ana LAN'dan ssh admin@192.168.99.2 ile switch yönetim arayüzüne ulaşabilmeli.
  5. 5nft list ruleset ile aktif kuralları doğrula. logread | grep vlan ile düşürülen paketleri izle.
